\subsection{Wiener--Hopf Factorisation on the Semi-Infinite Fermi Sea}
In \script{toda\_edge\_ba.py}, the ground-state density $\rho(p)$ and phase-shift dressing function
$F(p|p_2)$ on $[-q,q]$ satisfy
\begin{equation}
  2\pi h\,\rho(p) - \int_{-q}^q K(p-p')\,\rho(p')\,dp' = 1, \qquad
  2\pi h\,F(p|p_2) - \int_{-q}^q K(p-p')\,F(p'|p_2)\,dp' = -\Theta(p-p_2),
\end{equation}
where $h = \hbar_T = 3b\hbar_B$ ($a=1$, $b=e^{-1/2}$) and
\begin{equation}
  \Theta(D) = 2h\operatorname{Im}\ln\Gamma\!\left(1 + \frac{iD}{h}\right) + 2D\ln h, \qquad
  K(D) = \Theta'(D) = 2\operatorname{Re}\psi\!\left(1 + \frac{iD}{h}\right) + 2\ln h.
\end{equation}
In the Fermi-edge scaling limit $p = q + h x$ ($h\to 0$ at fixed $x$), the left edge $-q$ recedes
to $x = -2q/h \to -\infty$, so the Fermi sea occupies the negative half-line $x' \in (-\infty,0)$
and the integral operator becomes a Wiener--Hopf operator with kernel $K_0(x-x') = 2\operatorname{Re}\psi(1+i(x-x'))$.

\begin{theorem}[Wiener--Hopf factorisation and dressed momentum, \Lc, \scrshort]\label{thm:wh_du}
\begin{enumerate}
\item The Fourier symbol of $2\pi\delta(x) - K_0(x)$ is
  \begin{equation}\label{eq:wh_symbol}
    \Sigma_q(\omega) := \int_{-\infty}^\infty \big[2\pi\delta(x) - 2\operatorname{Re}\psi(1+ix)\big] e^{-i\omega x}\,dx
    = \frac{2\pi^2}{\sinh(\pi|\omega|)}e^{\pi|\omega|}
    = \Phi_{\mathrm{WH}}(i\omega)\,\Phi_{\mathrm{WH}}(-i\omega),
  \end{equation}
  where the canonical Wiener--Hopf factor analytic and non-vanishing in $\operatorname{Re} s > 0$ ($s = -i\omega$) is
  \begin{equation}\label{eq:phi_wh}
    \Phi_{\mathrm{WH}}(s) = \sqrt{\frac{2\pi}{s}}\,\Gamma\!\left(1 + \frac{s}{2\pi}\right)
    \exp\!\left(-\frac{s}{2\pi}\ln\frac{s}{2\pi e}\right), \qquad
    \widetilde{\Phi}_{\mathrm{WH}}(s) := \frac{\Phi_{\mathrm{WH}}(s)}{\sqrt{2\pi}} \underset{s\to+\infty}{\longrightarrow} 1.
  \end{equation}
\item Matching the homogeneous Wiener--Hopf edge density to the bulk classical arcsine asymptote
  $\rho_{\mathrm{cl}}(q - hz) \sim \frac{1}{2\pi a\sqrt{bhz}}$ ($z\to+\infty$) yields the exact
  Laplace transform of the scaled dressed momentum
  $\lim_{h\to 0} \frac{a\,[P_{\mathrm{dr}}(q+hu)-P_{\mathrm{dr}}(q)]}{\sqrt{bh}\,h} = \frac{4}{3}y(u)^{3/2}$
  for $u > 0$:
  \begin{equation}\label{eq:laplace_y}
    \int_0^\infty \left(\frac{4}{3}y(u)^{3/2}\right) e^{-su}\,du
    = \frac{\widetilde{\Phi}_{\mathrm{WH}}(s)}{s^2}
    = s^{-5/2}\,\Gamma\!\left(1 + \frac{s}{2\pi}\right)\exp\!\left(-\frac{s}{2\pi}\ln\frac{s}{2\pi e}\right).
  \end{equation}
\item Expanding~\eqref{eq:laplace_y} around the branch point $s = 0$ via
  $\ln\Gamma(1+\frac{s}{2\pi}) = -\gamma_E\frac{s}{2\pi} + \frac{\pi^2}{12}(\frac{s}{2\pi})^2 - \frac{\zeta(3)}{3}(\frac{s}{2\pi})^3 + \cdots$
  and inverting term-by-term via $\mathcal{L}^{-1}[s^{-\alpha}(\ln s)^k](u) = (-1)^k\partial_\alpha^k[u^{\alpha-1}/\Gamma(\alpha)]$
  gives as $u\to+\infty$:
  \begin{equation}\label{eq:y_of_u}
    y(u) - u = \frac{\ln(8\pi u) - 1}{2\pi}
    + \frac{12\ln(8\pi u) - \pi^2}{48\pi^2\,u}
    + \frac{-12\ln(8\pi u) + 6\zeta(3) + \pi^2 - 8}{96\pi^3\,u^2}
    + O(u^{-3}).
  \end{equation}
\item Inverting~\eqref{eq:y_of_u} for $\delta u(y) := u(y) - y$ as $y\to+\infty$, all
  $\frac{\ln y}{y}$, $\frac{\ln^2 y}{y^2}$, and $\frac{\ln y}{y^2}$ terms cancel identically
  ($d_1 = e_2 = e_1 = 0$), leaving the pure power-tail expansion after the leading logarithm:
  \begin{equation}\label{eq:du_exact}
    \boxed{\delta u(y) = -\frac{\ln y}{2\pi} + c_{\mathrm{edge}} + \frac{d_0}{y} + \frac{e_0}{y^2} + O(y^{-3}),}
  \end{equation}
  with exact analytical coefficients
  \begin{align}
    c_{\mathrm{edge}} &= \frac{1 - \ln(8\pi)}{2\pi} = -0.35398787697503518285\dots, \label{eq:cedge}\\
    d_0 &= \frac{1}{48} - \frac{1}{4\pi^2} = -0.00449696257725110953\dots, \label{eq:d0}\\
    e_0 &= \frac{4 + 6\zeta(3) - \pi^2}{96\pi^3} = +0.00045109717881779249\dots \label{eq:e0}
  \end{align}
\end{enumerate}
\end{theorem}

\subsection{Krein--Wiener--Hopf Convolution for the Two-Particle Dressed Phase}
We now derive the two-particle Fermi-edge dressed phase
\begin{equation}
  \Phi_q(u_1,u_2) := \lim_{h\to 0}\frac{\Phi(q+hu_1\,|\,q+hu_2) - \Phi(q+hu_1\,|\,q)}{h} \qquad (u_1 > u_2 > 0)
\end{equation}
directly from the Bethe integral equation, resolving Section~3.2 and Section~4 of
\texttt{independent\_phase2h\_audit.pdf} and Section~4 of \texttt{audit\_frames\_defects.pdf}.

\begin{theorem}[Krein convolution and large-$Y$ expansion of $R_0(Y,r)$, \Lc, \scrshort]\label{thm:krein_c1}
\begin{enumerate}
\item For $u_1, u_2 > 0$ outside the Fermi sea, the Wiener--Hopf projection onto $x > 0$ yields
  the exact double Laplace transform
  \begin{equation}\label{eq:double_laplace}
    \int_0^\infty du_1\,e^{-s_1 u_1}\int_0^\infty du_2\,e^{-s_2 u_2}\,\frac{\partial\Phi_q(u_1,u_2)}{\partial u_2}
    = 2\,\frac{\widetilde{\Phi}_{\mathrm{WH}}(s_1)\,\widetilde{\Phi}_{\mathrm{WH}}(s_2) - 1}{s_1 + s_2}.
  \end{equation}
  For $u_1 > u_2 > 0$ (where $\mathcal{L}^{-1}[(s_1+s_2)^{-1}] = \delta(u_1-u_2) = 0$), integrating
  with respect to $u_2$ from the edge $u_2 = 0$ ($\Phi_q(u_1,0) = 0$) gives the exact convolution
  \begin{equation}\label{eq:krein_conv}
    \boxed{\Phi_q(u_1,u_2) = 2\int_0^{u_2} \psi_-(u_1 - w)\,\widetilde{\Psi}_-(u_2 - w)\,dw
    = 4\int_0^{r^2 Y} \psi_-\big(u(Y) - u(r^2 Y) + u(\eta)\big)\,\sqrt{\eta}\,d\eta,}
  \end{equation}
  where $\widetilde{\Psi}_-(u) = \frac{d}{du}\big(\frac{4}{3}y(u)^{3/2}\big) = 2\sqrt{y(u)}\,y'(u)$,
  $\psi_-(u) = \widetilde{\Psi}_-'(u)$, and in the second equality we changed integration variable to
  the dressed momentum $v = u_2 - w = u(\eta)$ with $\eta \in [0, r^2 Y]$.
\item Setting $\eta = Yt^2$ ($t \in [0,r]$) and $\eta_0(t) := Y(1-r^2+t^2) =: Y\sigma(t)$, the argument
  of $\psi_-$ in~\eqref{eq:krein_conv} is $u(\eta_0(t)) + \Delta(Y,t)$, where $\Delta(Y,t)$ is the
  balanced 4-term coboundary
  \begin{equation}\label{eq:delta_coboundary}
    \Delta(Y,t) = \delta u(Y) - \delta u(r^2 Y) + \delta u(Yt^2) - \delta u\big(Y\sigma(t)\big)
    = \underbrace{-\frac{1}{2\pi}\ln\frac{t^2}{r^2\sigma(t)}}_{\Delta_0(t)}
    + \frac{d_0}{Y}\underbrace{\left(1 - \frac{1}{r^2} + \frac{1}{t^2} - \frac{1}{\sigma(t)}\right)}_{\Delta_1(t)}
    + O(Y^{-2}),
  \end{equation}
  in which both $-\frac{\ln Y}{2\pi}$ and $c_{\mathrm{edge}}$ cancel identically. Moreover, because
  $d_1 = 0$, $\psi_-(u(\eta)) = \big(\frac{1}{u'(\eta)}\frac{d}{d\eta}\big)^2(\frac{4}{3}\eta^{3/2})
  = \eta^{-1/2}\big[1 - (2d_0 + \frac{3}{4\pi^2})\eta^{-2} + O(\eta^{-3})\big]$ contains neither
  $\ln\eta$ nor $c_{\mathrm{edge}}$. Consequently, $\Phi_q(u(Y),u(r^2 Y))$ has \textbf{no} $\ln Y$ or
  $c_{\mathrm{edge}}$ terms at $O(Y^0)$ or $O(Y^{-1})$: the $\frac{1-r^2}{4\pi^2 r}\frac{\ln^2 Y}{Y}$
  and $\frac{\ln Y}{Y}$ terms in the bare classical Hessian $Q = \frac{(\delta u_2 - r^2\delta u_1)^2}{r(1-r^2)Y}$
  are cancelled identically by opposite terms in $d\Phi_{\mathrm{loop}}(Y,r) = \Phi_q(u_1,u_2) - \Phi_{\mathrm{cl}}(u_1,u_2)$.
\item Under the rationalising substitution $s = \frac{t}{\sqrt{1-r^2+t^2}} \in [0,r]$
  ($\Delta_0(t) = -\frac{1}{\pi}\ln\frac{s}{r}$, $\frac{t^2\,dt}{\sigma(t)^{3/2}} = (\frac{1}{1-s^2}-1)\,ds$),
  the $O(Y^0)$ integral gives in one line the analytical proof of the leading residual:
  \begin{equation}\label{eq:r0_inf_proof}
    \boxed{\lim_{Y\to\infty} R_0(Y,r) = -4\int_0^r \Delta_0(t)\frac{t^2\,dt}{(1-r^2+t^2)^{3/2}}
    = \frac{4}{\pi}\int_0^r \ln\!\left(\frac{s}{r}\right)\left(\frac{1}{1-s^2} - 1\right)ds
    = -\frac{4}{\pi}\big(\chi_2(r) - r\big).}
  \end{equation}
\item At $O(1/Y)$ (two loops in the Boussinesq/KdV scaling limit), the Wiener--Hopf integral evaluates
  in closed elementary form to
  \begin{equation}\label{eq:c1_exact}
    \boxed{R_0(Y,r) = -\frac{4}{\pi}\big(\chi_2(r)-r\big) + \frac{c_1(r)}{Y} + O(Y^{-2}), \qquad
    c_1(r) = -\frac{r^3}{6(1-r^2)} - 2d_0\left(\frac{1-r^2}{r^2}\ln\frac{1-r}{1+r} + \frac{2}{r}\right),}
  \end{equation}
  where $d_0 = \frac{1}{48} - \frac{1}{4\pi^2}$. Relative to the bare two-body Gamma-ratio Stirling
  coefficient $c_1^\Gamma(r) = -\frac{r^2}{6(1-r^2)}$ of \texttt{smatrix\_ansatz.tex}, the $\frac{1}{1-r}$
  threshold pole cancels identically in the collective Fermi-sea dressing shift:
  \begin{equation}\label{eq:dc1_exact}
    \Delta c_1(r) := c_1(r) - c_1^\Gamma(r)
    = \frac{r^2}{6(1+r)} - 2d_0\left(\frac{1-r^2}{r^2}\ln\frac{1-r}{1+r} + \frac{2}{r}\right), \qquad
    \lim_{r\to 1^-} \Delta c_1(r) = \frac{1}{\pi^2}.
  \end{equation}
\end{enumerate}
\end{theorem}

\subsection{Exchange-Antisymmetry Criterion and the Subtracted Kernel $\widetilde{K}_L(z)$}
We now resolve Section~2 of \texttt{independent\_phase2h\_audit.pdf} and Section~2 of

\begin{theorem}[Inversion-antisymmetry and 4-orientation identities, \Lc, \scrshort]\label{thm:inversion_orientation}
\begin{enumerate}
\item Under particle exchange $(1 \leftrightarrow 2)$ or KP-vertex swap $(p \leftrightarrow q)$,
  every root-ratio argument $z = w_r(\alpha_1,\epsilon_1)/w_s(\alpha_2,\epsilon_2)$ inverts: $z \mapsto 1/z$.
  Consequently, $\mathcal{F}_K(p,q) = -\mathcal{F}_K(q,p)$ and
  $\mathcal{I}_{SS}[K](1,2) = -\mathcal{I}_{SS}[K](2,1)$ hold universally if and only if $K(z)$ is
  \textbf{both odd ($K(-z) = -K(z)$) and inversion-antisymmetric ($K(1/z) = -K(z)$)}.
\item For the one-loop kernel $K_L(z) = -\frac{1}{\pi}\chi_2(z) - \frac{1}{2\pi}\ln|z|\ln\big|\frac{1-z}{1+z}\big|$,
  Euler's identity $\chi_2(z) + \chi_2(1/z) = \frac{\pi^2}{4}\operatorname{sgn}(\operatorname{Re} z)$ yields the
  inversion anomaly $K_L(z) + K_L(1/z) = -\frac{\pi}{4}\operatorname{sgn}(\operatorname{Re} z)$.
  Defining the inversion-antisymmetrised kernel
  \begin{equation}\label{eq:ktilde_L}
    \boxed{\widetilde{K}_L(z) := \frac{K_L(z) - K_L(1/z)}{2} = K_L(z) + \frac{\pi}{8}\operatorname{sgn}(\operatorname{Re} z)}
  \end{equation}
  makes $\widetilde{K}_L(-z) = -\widetilde{K}_L(z)$ and $\widetilde{K}_L(1/z) = -\widetilde{K}_L(z)$

  $\mathcal{F}_{\widetilde{K}_L}(p,q) = -\mathcal{F}_{\widetilde{K}_L}(q,p)$ and
  $\mathcal{I}_{SS}[\widetilde{K}_L](1,2) = -\mathcal{I}_{SS}[\widetilde{K}_L](2,1)$
  (with $\mathcal{I}_{SS}[\widetilde{K}_L](\alpha,\epsilon;\alpha,\epsilon) = 0$).
\item Because $\cot(\phi_r/2)$ has the same signs $(+,-,+)$ as $\tan(\phi_r/2)$ for all $\alpha \in (0,\pi/3)$,
  $\sum_{r=1}^3 c_r\operatorname{sgn}(w_r(\alpha,\epsilon)) = -2$ for \textbf{both} $\epsilon = \pm 1$.
  Across all four orientations $(\epsilon_1,\epsilon_2) \in \{+1,-1\}^2$ and all three defect regimes

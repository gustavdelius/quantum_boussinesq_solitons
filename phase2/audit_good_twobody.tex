\begin{equation}

\end{equation}

\end{equation}
For two solitons, the Hirota interaction coefficient $\tilde{A}_{12} = -\tilde{\mathcal{P}}(K_1-K_2,W_1-W_2)/\tilde{\mathcal{P}}(K_1+K_2,W_1+W_2)$ has $b^2/d$ cancelling between numerator and denominator, so $\tilde{A}_{12}(\alpha_1,\alpha_2) = A_{12}(\alpha_1,\alpha_2)$ at fixed angles $\alpha_j$. \end{equation}

\subsection{Overtaking Limits ($A_{12} \to 0^+$) and the False $|\alpha_1-\alpha_2|=\pi/3$ Line}

\begin{enumerate}
\item \textbf{Light equal-velocity overtaking threshold ($\alpha_1 \in (0,\pi/3)$, $\alpha_2 = \alpha_1 + h$, $h \to 0^+$)}: Here $A_{12}(h) = c_0(\alpha_1)h^2 + O(h^3) \to 0^+$, so $\ln A_{12}(h) \sim 2\ln h \to -\infty$ and the individual Darboux shifts $\Delta X'_j \sim \mp\frac{2}{k_1}\ln h$ diverge. However, because the two solitons approach identical parameters ($\alpha_2 \to \alpha_1$, $\epsilon_2 = \epsilon_1$), the prefactors in $R[p] = (p_2/k_2 - p_1/k_1)\ln A_{12} + O(h)$ and $K_\lambda = \frac{2}{3}(v_2-v_1)\ln A_{12} + O(h)$ vanish linearly in $h$:
\begin{equation}
K_\lambda(\alpha_1, \alpha_1+h; ++) = -\frac{4}{3}\sin\alpha_1\,h\ln h + O(h) \to 0, \qquad R[p](\alpha_1,\alpha_1+h; ++) = O(h\ln h) \to 0, \qquad J_{12} = O(h) \to 0.
\end{equation}
At $\alpha_1=0.4, h=10^{-12}$, $K_\lambda = +1.434298\times 10^{-11}$, matching $-\frac{4}{3}\sin(0.4)\,h\ln h = +1.434663\times 10^{-11}$ to $3.65\times 10^{-15} = O(h)$.
\item \textbf{Heavy overtaking Y-line ($\alpha_1 + \alpha_2 = 2\pi/3 - h$, $h \to 0^+$)}: Because $\alpha_1 + \alpha_2 = 2\pi/3$ cannot be reached with both $\alpha_j \in (0,\pi/3)$, at least one soliton must lie on the spectrally unstable heavy branch $\alpha > \pi/3$. At $\alpha_1=0.9$, $\alpha_2^* = 2\pi/3 - 0.9 \approx 1.194395 > \pi/3$, $A_{12}(h) \propto h \to 0^+$ while $v_2^* - v_1 = \cos\alpha_2^* - \cos\alpha_1 \approx -0.254266 \neq 0$, so $K_\lambda(h) \sim \frac{2}{3}(v_2^*-v_1)\ln h \to +\infty$ diverges logarithmically (increasing by $-\frac{2}{3}(v_2^*-v_1)\ln 100 \approx +0.779913$ per factor of $100$ in $h$). Because the heavy branch has tachyonic internal modes, \texttt{good\_twobody\_resonance.json} rightly sets \texttt{quantum\_phase: null}.
\item \textbf{Refutation of the false overtaking resonance $|\alpha_1 - \alpha_2| = \pi/3$}: At $(\alpha_1,\alpha_2) = (0.2, 0.2+\pi/3)$ with $\epsilon_1 = \epsilon_2 = +1$, $A_{12} = 0.35546547984906165\dots$, which is strictly positive and finite ($0 < A_{12} < 1$). The numerator of $A_{12}$ factors via $\cos 3\alpha_1 - \cos 3\alpha_2 = 2\sin\frac{3(\alpha_2-\alpha_1)}{2}\sin\frac{3(\alpha_1+\alpha_2)}{2}$, whose zeroes in the overtaking channel are $\alpha_2 - \alpha_1 = 0$ and $\alpha_1 + \alpha_2 = 2\pi/3$, \emph{not} $|\alpha_1-\alpha_2|=\pi/3$.
\end{enumerate}

\section{Failures Recorded}

In compliance with \texttt{AGENTS.md} (``Report failures honestly. \subsection{Historical Failures Documented in \texttt{good\_twobody.tex}}
\begin{enumerate}
\item \textbf{Legacy fixed-momentum counterterm vanishing claim}: Earlier exploratory Phase~2 notes claimed that $\Delta H = \lambda\hbar\int u^2\,\dd x$ produced zero shift of the two-soliton phase at fixed physical momenta. Direct scaling and spacetime integration refute this: $K_\lambda = 3\delta_0 + 4\sum_j P_j \Delta X'_j \neq 0$.
\item \textbf{Legacy overtaking resonance at $|\alpha_1-\alpha_2|=\pi/3$}: Earlier Phase~2 notes placed an overtaking resonance at $|\alpha_1-\alpha_2|=\pi/3$. Direct evaluation gives $A_{12}(0.2, 0.2+\pi/3; ++) \approx 0.355465 \neq 0,\infty$; the actual zeros/poles of $A_{12}$ are $\alpha_1+\alpha_2=\pi/3$ (head-on Y-resonance), $\alpha_1=\alpha_2$ (overtaking threshold), and $\alpha_1+\alpha_2=2\pi/3$ (heavy overtaking Y-line).
\item \textbf{Unjustified setting of $B_{12}=0$ and physical symmetric phase to zero}: Earlier notes set the unknown boundary-projection phase $B_{12}$ to zero without computing the projected two-soliton determinant or collective-radiation Schur complement. The audited package replaces those entries with explicit \texttt{null} values.
\end{enumerate}
